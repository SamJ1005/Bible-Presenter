//main.js
const {
  app,
  BrowserWindow,
  ipcMain,
  screen
} = require("electron");
const path = require("path");

let mainWin = null;
let presentationWin = null;

/* -----------------------------------------------------
   MAIN WINDOW
------------------------------------------------------ */
function createMainWindow() {
  mainWin = new BrowserWindow({
    width: 1200,
    height: 800,
    webPreferences: {
      preload: path.join(__dirname, "preload.js"),
      contextIsolation: true,
    },
  });

  mainWin.loadURL("http://localhost:5173");
}

/* -----------------------------------------------------
   SELECT BEST DISPLAY (Dynamic)
------------------------------------------------------ */
function getPresentationDisplay() {
  const displays = screen.getAllDisplays();

  // Prefer external monitor
  const external = displays.find(d => d.internal === false);
  if (external) return external;

  // Or any non-primary
  const nonPrimary = displays.find(d => !d.primary);
  if (nonPrimary) return nonPrimary;

  // Fallback -> laptop screen
  return displays[0];
}

/* -----------------------------------------------------
   MOVE PRESENTATION WINDOW TO BEST DISPLAY (LIVE)
------------------------------------------------------ */
function movePresentationToBestDisplay() {
  if (!presentationWin || presentationWin.isDestroyed()) return;

  const target = getPresentationDisplay();
  const {
    x,
    y,
    width,
    height
  } = target.bounds;

  // Disable fullscreen first
  presentationWin.setFullScreen(false);

  // Move to new display
  presentationWin.setBounds({
    x,
    y,
    width,
    height
  });

  // Re-apply fullscreen
  presentationWin.setFullScreen(true);
}
/* -----------------------------------------------------
   OPEN BLANK PRESENTATION (Black Screen)
------------------------------------------------------ */
function openBlankPresentation() {
  if (presentationWin && !presentationWin.isDestroyed()) return;

  const target = getPresentationDisplay();
  const {
    x,
    y,
    width,
    height
  } = target.bounds;

  presentationWin = new BrowserWindow({
    x,
    y,
    width,
    height,
    frame: false,
    fullscreen: false,
    show: false,
    backgroundColor: "#000000",
    autoHideMenuBar: true,
    webPreferences: {
      preload: path.join(__dirname, "preload.js"),
      contextIsolation: true,
    },
  });

  // Load blank presentation HTML
  presentationWin.loadFile(path.join(__dirname, "public", "presentation.html"));

  presentationWin.once("ready-to-show", () => {
    movePresentationToBestDisplay();
    presentationWin.show();
  });

  presentationWin.on("closed", () => {
    presentationWin = null;
  });
}

/* -----------------------------------------------------
   CLOSE PRESENTATION
------------------------------------------------------ */
function closePresentation() {
  if (presentationWin) {
    presentationWin.close();
    presentationWin = null;
  }
}

/* -----------------------------------------------------
   IPC HANDLERS
------------------------------------------------------ */
ipcMain.on("OPEN_BLANK_PRESENTATION", openBlankPresentation);
ipcMain.on("CLOSE_PRESENTATION", closePresentation);
ipcMain.on("SEND_PRESENTATION_DATA", (event, payload) => {
  if (presentationWin && !presentationWin.isDestroyed()) {
    presentationWin.webContents.send("presentation-update", payload);
  }
});
ipcMain.on("presentation-prev", () => {
  if (mainWin && !mainWin.isDestroyed()) {
    mainWin.webContents.send("ui-prev-verse");
  }
});

ipcMain.on("presentation-next", () => {
  if (mainWin && !mainWin.isDestroyed()) {
    mainWin.webContents.send("ui-next-verse");
  }
});
ipcMain.handle("PRESENTATION_STATUS", () => {
  return !!presentationWin && !presentationWin.isDestroyed();
});
/* -----------------------------------------------------
   APP READY
------------------------------------------------------ */
app.whenReady().then(() => {
  createMainWindow();

  // 🔥 Live monitor detection
  screen.on("display-added", () => {
    movePresentationToBestDisplay();
  });

  screen.on("display-removed", () => {
    movePresentationToBestDisplay();
  });

  screen.on("display-metrics-changed", () => {
    movePresentationToBestDisplay();
  });
  app.on('activate', () => {
    if (BrowserWindow.getAllWindows().length === 0) createMainWindow();
  });
});

//preload.js
const {
  contextBridge,
  ipcRenderer
} = require("electron");

/* -----------------------------------------------------
   API for MAIN UI (React)
------------------------------------------------------ */
contextBridge.exposeInMainWorld("electron", {
  // UI → Main
  openBlankPresentation: () => ipcRenderer.send("OPEN_BLANK_PRESENTATION"),
  closePresentation: () => ipcRenderer.send("CLOSE_PRESENTATION"),
  sendPresentation: (payload) => ipcRenderer.send("SEND_PRESENTATION_DATA", payload),
  getPresentationStatus: () => ipcRenderer.invoke("PRESENTATION_STATUS"),
  onPrev: (cb) => {
    const listener = (_e, data) => cb(data);
    ipcRenderer.on("ui-prev-verse", listener);
    return () => ipcRenderer.removeListener("ui-prev-verse", listener);
  },
  onNext: (cb) => {
    const listener = (_e, data) => cb(data);
    ipcRenderer.on("ui-next-verse", listener);
    return () => ipcRenderer.removeListener("ui-next-verse", listener);
  },
});
/* -----------------------------------------------------
   API for PRESENTATION WINDOW (Arrow Key Control)
------------------------------------------------------ */
contextBridge.exposeInMainWorld("api", {
  // Presentation → Main
  prevVerse: () => ipcRenderer.send("presentation-prev"),
  nextVerse: () => ipcRenderer.send("presentation-next"),
  closePresentation: () => ipcRenderer.send("CLOSE_PRESENTATION"),
});

/* -----------------------------------------------------
   Presentation window receives updates from UI
------------------------------------------------------ */
contextBridge.exposeInMainWorld("presentationAPI", {
  onShowVerse: (callback) => ipcRenderer.on("presentation-update", callback),
});

      document.addEventListener("keydown", (e) => {
        if (e.key === "ArrowLeft") window.api.prevVerse();
        if (e.key === "ArrowRight") window.api.nextVerse();
        if (e.key === "Escape") window.api.closePresentation();
      });

      function autoSize(el, maxSize) {
        if (!el) return;
        let size = maxSize || 48;
        el.style.fontSize = size + "px";
        const parent = el.parentElement;
        while (
          (el.scrollHeight > parent.clientHeight ||
            el.scrollWidth > parent.clientWidth) &&
          size > 20
        ) {
          size -= 2;
          el.style.fontSize = size + "px";
        }
      }

      // Debug helper so you can see the messages arriving
      function debugLog(...args) {
        // prints to presentation window console (open devtools to view)
        console.log("[presentation]", ...args);
      }

      window.presentationAPI.onShowVerse((_event, payload) => {
        debugLog("presentation-update payload:", payload);

        const tamil = document.getElementById("tamil");
        const english = document.getElementById("english");
        const index = document.getElementById("index");

        tamil.innerText = payload.tamilText || "";
        english.innerText = payload.englishText || "";
        index.innerText = payload.index || "";

        // autosize using provided sizes (or fallbacks)
        if (payload.tamilEnabled) autoSize(tamil, payload.tamilFontSize || 48);
        if (payload.englishEnabled)
          autoSize(english, payload.englishFontSize || 36);
      });

      // Optional: show a small marker so you know when the presentation page is live
      console.log("Presentation page loaded and listening for updates");
    </script>
  </body>
